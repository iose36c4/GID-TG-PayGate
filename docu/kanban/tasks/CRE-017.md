# [[CRE-017]] Facturación Creador: Detalle, PDF, CAE, Email, Descarga, Historial

## Descripción

```blade
function viewInvoice(id) {
    fetch(`{{ route('creadores.invoices.show', ['channel' => $channel->id, 'invoice' => ':id']) }}`.replace(':id', id))

    document.getElementById('invoice-modal').classList.remove('flex');
    document.getElementById('invoice-modal-content').innerHTML = '';
}

    window.location.href = `{{ route('creadores.invoices.download', ['channel' => $channel->id, 'invoice' => ':id']) }}`.replace(':id', id);
}

function resendInvoice(id) {
    fetch(`{{ route('creadores.invoices.resend', ['channel' => $channel->id, 'invoice' => ':id']) }}`.replace(':id', id), {
        method: 'POST',
        headers: {
            'Content-Type': 'application/json',
            'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content
        }

    .then(data => {

            alert('Factura reenviada correctamente');
        } else {
            alert('Error: ' + data.message);
        }
    });
}
```

```php
// app/Domains/Creadores/Http/Controllers/InvoiceController.php
namespace App\Domains\Creadores\Http\Controllers;

use App\Models\{ChannelPago, Invoice};

class InvoiceController extends Controller
{
    public function index(ChannelPago $channel)
    {
        $this->authorize('view', $channel);
        
        $invoices = Invoice::where('channel_pago_id', $channel->id)
            ->when(request('status'), fn($q) => $q->where('status', request('status')))
            ->when(request('type'), fn($q) => $q->where('type', request('type')))
            ->when(request('date_from'), fn($q) => $q->whereDate('created_at', '>=', request('date_from')))
            ->when(request('date_to'), fn($q) => $q->whereDate('created_at', '<=', request('date_to')))
            ->latest()
            ->paginate(20)
            ->withQueryString();

        return view('domains.creadores.channels.invoices.index', compact('channel', 'invoices'));
    }

    public function show(ChannelPago $channel, Invoice $invoice)
    {
        $this->authorize('view', $channel);
        $invoice->load(['channel', 'receiver']);
        return view('domains.creadores.channels.invoices.show', compact('channel', 'invoice'));
    }

    public function download(ChannelPago $channel, Invoice $invoice)
    {
        $this->authorize('view', $channel);
        
        if (!$invoice->pdf_path || !\Storage::disk('local')->exists($invoice->pdf_path)) {
            abort(404, 'PDF no disponible');
        }

        return \Storage::disk('local')->download($invoice->pdf_path, "factura-{$invoice->number}.pdf");
    }

    public function resend(ChannelPago $channel, Invoice $invoice)
    {
        $this->authorize('manage', $channel);
        
        if ($invoice->status !== 'authorized') {
            return response()->json(['success' => false, 'message' => 'Solo facturas autorizadas pueden reenviarse'], 422);
        }

        $creator = $invoice->receiver;
        \Mail::to($creator->email)->send(new \App\Mail\CreatorInvoiceMail(
            $invoice->payout, // necesita relación payout
            $invoice
        ));

        return response()->json(['success' => true, 'message' => 'Factura reenviada']);
    }
}
```

## Criterios de Aceptación
- [ ] Listado facturas con filtros: estado (authorized/rejected/cancelled), tipo (A/B/E), fecha desde/hasta
- [ ] Tabla: número, fecha, tipo (badge color), total, estado (badge), CAE, acciones
- [ ] Paginación 20 por página, orden fecha desc
- [ ] Modal detalle: PDF embebido (iframe), datos completos emisor/receptor, detalle items, totales, retenciones, QR AFIP, PDF417
- [ ] Descarga PDF directo (Storage::download)
- [ ] Reenviar email: solo facturas authorized, botón en modal y tabla
- [ ] Historial completo: todas las facturas del canal desde creación
- [ ] Exportar CSV: con filtros aplicados, columnas estándar
- [ ] Estados: authorized (verde), rejected (rojo), cancelled (amarillo), draft (gris)
- [ ] Performance: eager loading relations, paginación eficiente

## Notas Técnicas
- PDF almacenado en `storage/app/invoices/{number}.pdf`
- Modal usa iframe o object tag para visualizar PDF inline
- QR AFIP generado al autorizar: `https://www.afip.gob.ar/fe/qr/?p={base64}`
- PDF417 generado con biblioteca `bacon/bacon-qr-code` o similar
- Email reenvío usa misma clase `CreatorInvoiceMail` que envío original
- Índices BD: channel_pago_id + status + created_at para queries rápidas

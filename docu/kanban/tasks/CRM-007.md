# [[CRM-007]] Reportes: Volumen, Tiempo Respuesta, Resolución, CSAT, Backlog Aging

## Descripción
Implementar panel de reportes en CRM: volumen de tickets, tiempo de respuesta/resolución, CSAT, aging de backlog, exportación CSV/PDF.

## Código de Ejemplo
```php
// app/Domains/Staff/Http/Controllers/Crm/ReportController.php
namespace App\Domains\Staff\Http\Controllers\Crm;

class ReportController extends Controller
{
    public function index()
    {
        return view('domains.staff.crm.reports.index');
    }

    public function volume(ReportRequest $request)
    {
        $data = $this->getVolumeData($request);
        return view('domains.staff.crm.reports.volume', compact('data'));
    }

    public function responseTime(ReportRequest $request)
    {
        $data = $this->getResponseTimeData($request);
        return view('domains.staff.crm.reports.response-time', compact('data'));
    }

    public function resolutionTime(ReportRequest $request)
    {
        $data = $this->getResolutionTimeData($request);
        return view('domains.staff.crm.reports.resolution-time', compact('data'));
    }

    public function csat(ReportRequest $request)
    {
        $data = $this->getCSATData($request);
        return view('domains.staff.crm.reports.csat', compact('data'));
    }

    public function backlogAging(ReportRequest $request)
    {
        $data = $this->getBacklogAgingData($request);
        return view('domains.staff.crm.reports.backlog-aging', compact('data'));
    }

    public function export(ReportExportRequest $request)
    {
        return \Maatwebsite\Excel\Facades\Excel::download(
            new \App\Exports\CrmReportExport($request->validated()),
            "crm-report-{$request->type}-{$request->date_from}-to-{$request->date_to}.xlsx"
        );
    }

    private function getVolumeData(ReportRequest $request): array
    {
        $query = Ticket::query();
        
        if ($request->date_from) $query->whereDate('created_at', '>=', $request->date_from);
        if ($request->date_to) $query->whereDate('created_at', '<=', $request->date_to);

        $byDay = $query->clone()
            ->selectRaw('DATE(created_at) as date, count(*) as count')
            ->groupBy('date')
            ->orderBy('date')
            ->get();

        $byStatus = Ticket::selectRaw('status, count(*) as count')
            ->groupBy('status')
            ->get();

        $byPriority = Ticket::selectRaw('priority, count(*) as count')
            ->groupBy('priority')
            ->get();

        $byCategory = Ticket::selectRaw('category_id, count(*) as count')
            ->groupBy('category_id')
            ->with('category')
            ->get();

        return compact('byDay', 'byStatus', 'byPriority', 'byCategory');
    }

    private function getResponseTimeData(ReportRequest $request): array
    {
        $tickets = Ticket::whereNotNull('first_response_at')
            ->when($request->date_from, fn($q) => $q->whereDate('created_at', '>=', $request->date_from))
            ->when($request->date_to, fn($q) => $q->whereDate('created_at', '<=', $request->date_to))
            ->get();

        $byPriority = $tickets->groupBy('priority')
            ->map(fn($group) => round($group->avg(fn($t) => $t->created_at->diffInMinutes($t->first_response_at)), 1));

        $byCategory = $tickets->groupBy('category_id')
            ->map(fn($group) => round($group->avg(fn($t) => $t->created_at->diffInMinutes($t->first_response_at)), 1));

        $byAssignee = $tickets->whereNotNull('assigned_to')
            ->groupBy('assigned_to')
            ->map(fn($group) => round($group->avg(fn($t) => $t->created_at->diffInMinutes($t->first_response_at)), 1));

        $slaCompliance = $tickets->whereNotNull('sla_deadline')
            ->whereNotNull('first_response_at')
            ->where('first_response_at', '<=', 'sla_deadline')
            ->count() / max(1, $tickets->whereNotNull('sla_deadline')->count()) * 100;

        return compact('byPriority', 'byCategory', 'byAssignee', 'slaCompliance');
    }

    private function getResolutionTimeData(ReportRequest $request): array
    {
        $tickets = Ticket::where('status', 'closed')
            ->whereNotNull('closed_at')
            ->when($request->date_from, fn($q) => $q->whereDate('created_at', '>=', $request->date_from))
            ->when($request->date_to, fn($q) => $q->whereDate('created_at', '<=', $request->date_to))
            ->get();

        $avgResolutionHours = $tickets->avg(fn($t) => $t->created_at->diffInHours($t->closed_at));
        $medianResolutionHours = $this->calculateMedian($tickets->map(fn($t) => $t->created_at->diffInHours($t->closed_at)));

        $byPriority = $tickets->groupBy('priority')
            ->map(fn($group) => round($group->avg(fn($t) => $t->created_at->diffInHours($t->closed_at)), 1));

        $byCategory = $tickets->groupBy('category_id')
            ->map(fn($group) => round($group->avg(fn($t) => $t->created_at->diffInHours($t->closed_at)), 1));

        return compact('avgResolutionHours', 'medianResolutionHours', 'byPriority', 'byCategory');
    }

    private function getCSATData(ReportRequest $request): array
    {
        $ratings = TicketRating::when($request->date_from, fn($q) => $q->whereDate('created_at', '>=', $request->date_from))
            ->when($request->date_to, fn($q) => $q->whereDate('created_at', '<=', $request->date_to))
            ->get();

        $avgRating = $ratings->avg('rating');
        $nps = $this->calculateNPS($ratings);
        $byAgent = TicketRating::with('ticket.assignee')
            ->where('created_at', '>=', $request->date_from ?? now()->subMonth())
            ->get()
            ->groupBy('ticket.assignee_id')
            ->map(fn($group) => [
                'avg_rating' => round($group->avg('rating'), 2),
                'count' => $group->count(),
            ]);

        return compact('avgRating', 'nps', 'byAgent');
    }

    private function getBacklogAgingData(ReportRequest $request): array
    {
        $openTickets = Ticket::whereIn('status', ['open', 'pending', 'waiting'])->get();

        $buckets = [
            '0-24h' => 0,
            '24-48h' => 0,
            '48-72h' => 0,
            '72h-1w' => 0,
            '1w-2w' => 0,
            '2w-1m' => 0,
            '1m+' => 0,
        ];

        foreach ($openTickets as $ticket) {
            $hours = $ticket->created_at->diffInHours(now());
            if ($hours <= 24) $buckets['0-24h']++;
            elseif ($hours <= 48) $buckets['24-48h']++;
            elseif ($hours <= 72) $buckets['48-72h']++;
            elseif ($hours <= 168) $buckets['72h-1w']++;
            elseif ($hours <= 336) $buckets['1w-2w']++;
            elseif ($hours <= 720) $buckets['2w-1m']++;
            else $buckets['1m+']++;
        }

        $byAssignee = Ticket::whereIn('status', ['open', 'pending', 'waiting'])
            ->whereNotNull('assigned_to')
            ->with('assignee')
            ->get()
            ->groupBy('assigned_to')
            ->map(fn($group) => [
                'count' => $group->count(),
                'avg_age_hours' => round($group->avg(fn($t) => $t->created_at->diffInHours(now())), 1),
                'oldest' => $group->max('created_at'),
            ]);

        return compact('buckets', 'byAssignee');
    }

    private function calculateMedian(array $values): float
    {
        sort($values);
        $count = count($values);
        if ($count === 0) return 0;
        $middle = floor($count / 2);
        if ($count % 2 === 0) {
            return ($values[$middle - 1] + $values[$middle]) / 2;
        }
        return $values[$middle];
    }

    private function calculateNPS(Collection $ratings): float
    {
        $total = $ratings->count();
        if ($total === 0) return 0;
        $promoters = $ratings->where('rating', '>=', 9)->count();
        $detractors = $ratings->where('rating', '<=', 6)->count();
        return round((($promoters - $detractors) / $total) * 100, 1);
    }
}
```

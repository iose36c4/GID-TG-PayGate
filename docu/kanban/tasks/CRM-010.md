# [[CRM-010]] Satisfacción: CSAT Survey Post-Cierre, NPS Trimestral

## Descripción
Implementar sistema de satisfacción del cliente en CRM: encuesta CSAT post-cierre de ticket, NPS trimestral, análisis de tendencias.

## Código de Ejemplo
```php
// app/Models/TicketRating.php
class TicketRating extends Model
{
    protected $fillable = [
        'ticket_id', 'user_id', 'rating', 'comment',
    ];

    protected $casts = [
        'rating' => 'integer',
    ];

    public function ticket() { return $this->belongsTo(Ticket::class); }
    public function user() { return $this->belongsTo(User::class); }
}
```

```php
// app/Domains/Staff/Http/Controllers/Crm/SatisfactionController.php
namespace App\Domains\Staff\Http\Controllers\Crm;

class SatisfactionController extends Controller
{
    public function index()
    {
        $ratings = TicketRating::with(['ticket', 'user'])
            ->when(request('date_from'), fn($q) => $q->whereDate('created_at', '>=', request('date_from')))
            ->when(request('date_to'), fn($q) => $q->whereDate('created_at', '<=', request('date_to')))
            ->latest()
            ->paginate(30);

        $stats = [
            'avg_rating' => TicketRating::avg('rating'),
            'total_ratings' => TicketRating::count(),
            'distribution' => TicketRating::selectRaw('rating, count(*) as count')
                ->groupBy('rating')
                ->pluck('count', 'rating'),
            'nps' => $this->calculateNPS(),
            'by_agent' => TicketRating::with('ticket.assignee')
                ->where('created_at', '>=', now()->subMonth())
                ->get()
                ->groupBy('ticket.assignee_id')
                ->map(fn($group) => [
                    'avg_rating' => round($group->avg('rating'), 2),
                    'count' => $group->count(),
                ]),
        );

        return view('domains.staff.crm.satisfaction.index', compact('ratings', 'stats'));
    }

    public function store(TicketRatingRequest $request, Ticket $ticket)
    {
        $rating = $ticket->ratings()->updateOrCreate(
            ['user_id' => auth()->id()],
            ['rating' => $request->rating, 'comment' => $request->comment]
        );

        // Si rating <= 3, crear alerta para supervisor
        if ($request->rating <= 3) {
            app(\App\Services\TelegramService::class)->notifyLowCSAT($ticket, $request->rating);
        }

        return response()->json(['success' => true, 'message' => 'Gracias por tu feedback']);
    }

    public function nps()
    {
        $nps = $this->calculateNPS();
        $history = NPSHistory::latest()->take(12)->get();

        return view('domains.staff.crm.satisfaction.nps', compact('nps', 'history'));
    }

    private function calculateNPS(): float
    {
        $ratings = TicketRating::where('created_at', '>=', now()->subMonths(3))->get();
        if ($ratings->isEmpty()) return 0;

        $promoters = $ratings->where('rating', '>=', 9)->count();
        $detractors = $ratings->where('rating', '<=', 6)->count();
        $total = $ratings->count();

        return round((($promoters - $detractors) / $total) * 100, 1);
    }
}
```

```blade
<script>
let selectedNPS = null;
function selectNPS(value) {
    selectedNPS = value;
    document.querySelectorAll('.nps-btn').forEach(btn => {
        btn.classList.remove('border-accent-primary', 'bg-accent-primary', 'text-white');
        if (parseInt(btn.dataset.value) === value) {
            btn.classList.add('border-accent-primary', 'bg-accent-primary', 'text-white');
        }
    }
    document.getElementById('comment-section').classList.toggle('hidden', value > 8);
}
function submitNPS() {
    if (selectedNPS === null) return alert('Selecciona una puntuación');
    fetch('{{ route("crm.satisfaction.nps.store") }}', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'X-CSRF-TOKEN': document.querySelector('meta[name="csrf-token"]').content },
        body: JSON.stringify({ score: selectedNPS, comment: document.getElementById('nps-comment')?.value })
    }).then(r => r.json()).then(data => {
        alert('Gracias por tu feedback');
        closeNPSModal();
    });
}
function closeNPSModal() { document.getElementById('nps-modal').classList.add('hidden'); }
</script>
@endpush
```

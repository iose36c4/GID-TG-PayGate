```php
// app/Http/Controllers/InstallController.php - saveAdmin
public function saveAdmin(AdminStoreRequest $request)
{
    $user = User::create([
        'name' => $request->name,
        'email' => $request->email,
        'password' => Hash::make($request->password),
        'role' => 'admin',
        'email_verified_at' => now(),
        'two_factor_confirmed_at' => $request->boolean('enable_2fa') ? now() : null,
        'two_factor_secret' => $request->boolean('enable_2fa') ? encrypt('placeholder') : null,
        'two_factor_recovery_codes' => $request->boolean('enable_2fa') ? encrypt(json_encode([])) : null,
    ]);

    $user->assignRole('admin');

    if ($request->boolean('enable_2fa')) {
        $user->update([
            'two_factor_secret' => encrypt('placeholder'),
            'two_factor_recovery_codes' => encrypt(json_encode([])),
        ]);
    }

    return redirect()->route('install.payment-gateways')
        ->with('success', 'Administrador creado correctamente');
}
```

```php
// app/Http/Requests/Install/AdminStoreRequest.php
class AdminStoreRequest extends FormRequest
{
    public function rules(): array
    {
        return [
            'name' => 'required|string|max:255',
            'email' => 'required|email|max:255|unique:users,email',
            'password' => 'required|string|min:8|confirmed',
            'enable_2fa' => 'boolean',
        ];
    }

    public function messages(): array
    {
        return [
            'email.unique' => 'Este email ya está registrado.',
            'password.min' => 'La contraseña debe tener al menos 8 caracteres.',
            'password.confirmed' => 'Las contraseñas no coinciden.',
        };
    }
}
```

## Criterios de Aceptación

- [ ] Validaciones: email único, password min 8 chars, confirmed
- [ ] Usuario creado con role=admin, email_verified_at=now()

- [ ] 2FA opcional: si checked, genera secret + recovery codes, two_factor_confirmed_at=now

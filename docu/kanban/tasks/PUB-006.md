# [[PUB-006]] Pasarela Pago Integration: Stripe/MercadoPago/PayPal (Abstracción + Webhooks)

## Descripción
Implementar capa de abstracción para pasarelas de pago: interfaz común `PaymentGatewayInterface`, implementaciones Stripe, MercadoPago, PayPal, factory para selección, manejo webhooks unificado.

## Código de Ejemplo
```php
// app/Domains/Public/Contracts/PaymentGatewayInterface.php
namespace App\Domains\Public\Contracts;

interface PaymentGatewayInterface
{
    public function createCheckout(array $data): array; // returns ['url', 'session_id', 'external_reference']
    public function verifyPayment(string $reference): array; // returns ['status', 'amount', 'metadata']
    public function refundPayment(string $paymentId, float $amount = null): array;
    public function getSupportedCurrencies(): array;
    public function getGatewayName(): string;
}
```

```php
// app/Domains/Public/Services/Gateways/MercadoPagoGateway.php
namespace App\Domains\Public\Services\Gateways;

use App\Domains\Public\Contracts\PaymentGatewayInterface;
use MercadoPago\Client\Preference\PreferenceClient;
use MercadoPago\MercadoPagoConfig;

class MercadoPagoGateway implements PaymentGatewayInterface
{
    public function __construct(
        protected string $accessToken,
        protected string $publicKey,
        protected string $mode = 'sandbox'
    ) {
        MercadoPagoConfig::setAccessToken($this->accessToken);
    }

    public function createCheckout(array $data): array
    {
        $client = new PreferenceClient();
        $preference = $client->create([
            'items' => [[
                'title' => $data['title'],
                'quantity' => 1,
                'currency_id' => $data['currency'] ?? 'ARS',
                'unit_price' => (float) $data['amount'],
            ]],
            'payer' => [
                'email' => $data['payer_email'],
                'name' => $data['payer_name'] ?? '',
                'surname' => $data['payer_surname'] ?? '',
            ],
            'back_urls' => [
                'success' => $data['success_url'],
                'failure' => $data['failure_url'],
                'pending' => $data['pending_url'],
            ],
            'auto_return' => 'approved',
            'notification_url' => $data['webhook_url'],
            'external_reference' => $data['external_reference'],
        ]);

        return [
            'url' => $preference['init_point'],
            'session_id' => $preference['id'],
            'external_reference' => $data['external_reference'],
        ];
    }

    public function verifyPayment(string $reference): array
    {
        $client = new PaymentClient();
        $payments = $client->search(['external_reference' => $reference]);
        
        $payment = $payments['results'][0] ?? null;
        if (!$payment) return ['status' => 'not_found'];

        return [
            'status' => $payment['status'], // approved, pending, rejected
            'amount' => $payment['transaction_amount'],
            'payment_id' => $payment['id'],
            'payment_method' => $payment['payment_method_id'],
            'installments' => $payment['installments'] ?? 1,
        ];
    }

    public function getGatewayName(): string { return 'mercadopago'; }
    public function getSupportedCurrencies(): array { return ['ARS', 'USD', 'BRL', 'MXN', 'COP', 'PEN', 'CLP', 'UYU']; }
}
```

```php
// app/Domains/Public/Services/PaymentGatewayFactory.php
class PaymentGatewayFactory
{
    public static function create(string $gateway, array $config): PaymentGatewayInterface
    {
        return match($gateway) {
            'mercadopago' => new MercadoPagoGateway(
                $config['access_token'],
                $config['public_key'],
                $config['mode'] ?? 'sandbox'
            ),
            'stripe' => new StripeGateway(
                $config['secret_key'],
                $config['publishable_key'],
                $config['webhook_secret']
            ),
            'paypal' => new PayPalGateway(
                $config['client_id'],
                $config['client_secret'],
                $config['mode'] ?? 'sandbox'
            ),
            default => throw new \InvalidArgumentException("Gateway no soportado: $gateway"),
        };
    }
}
```

```php
// app/Http/Controllers/Webhook/MercadoPagoWebhookController.php
class MercadoPagoWebhookController extends Controller
{
    public function handle(Request $request, ChannelPago $channel)
    {
        $this->validateSignature($request, $channel);
        
        $payload = $request->all();
        $topic = $payload['topic'] ?? '';
        
        if ($topic === 'payment') {
            $paymentId = $payload['data']['id'];
            $this->processPayment($paymentId, $channel);
        }

        return response()->json(['status' => 'ok']);
    }

    private function validateSignature(Request $request, ChannelPago $channel): void
    {
        $signature = $request->header('x-signature');
        $requestId = $request->header('x-request-id');
        
        if ($signature && $channel->mercadopago_client_secret) {
            $expected = hash_hmac('sha256', $requestId . '|' . $request->getContent(), 
                decrypt($channel->mercadopago_client_secret));
            
            if (!hash_equals($expected, $signature)) {
                abort(401, 'Firma inválida');
            }
        }
    }

    private function processPayment(string $paymentId, ChannelPago $channel): void
    {
        $mpService = app(\App\Services\MercadoPagoService::class);
        $payment = $mpService->getPayment($paymentId);
        
        $subscription = Subscription::where('external_reference', $payment['external_reference'])->first();
        
        if ($subscription && $payment['status'] === 'approved') {
            $subscription->update([
                'status' => 'active',
                'activated_at' => now(),
                'renews_at' => now()->addMonth(),
                'mercadopago_payment_id' => $payment['id'],
            ]);

            // Generar invite link + notificaciones
            app(\App\Services\TelegramService::class)->sendInviteLink($subscription);
        }
    }
}
```

## Criterios de Aceptación
- [ ] `PaymentGatewayInterface` con métodos: createCheckout, verifyPayment, refundPayment, getSupportedCurrencies
- [ ] 3 implementaciones: MercadoPagoGateway, StripeGateway, PayPalGateway
- [ ] PaymentGatewayFactory para instanciar según configuración
- [ ] Rutas webhook separadas por gateway con validación firma
- [ ] MercadoPago: HMAC SHA256 validation (x-signature header)
- [ ] Stripe: webhook signature verification (stripe-signature header)
- [ ] PayPal: webhook verification (paypal-signature header)
- [ ] Webhook idempotencia: external_reference único, procesar solo una vez
- [ ] Logging webhook: request/response completo para debugging
- [ ] Testing: mock webhooks, sandbox testing

## Notas Técnicas
- Interface permite agregar nuevos gateways sin romper código existente
- Factory pattern para instanciar gateway según config canal
- Webhooks: idempotencia crítica (external_reference único)
- MercadoPago: HMAC SHA256 (x-signature header)
- Stripe: stripe-signature header verification
- PayPal: webhook verification via certificates
- Queue jobs para procesamiento asíncrono webhooks
